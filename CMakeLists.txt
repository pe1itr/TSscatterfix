cmake_minimum_required(VERSION 3.16)
project(scatterfix VERSION 1.0 LANGUAGES C)
set(SCATTERFIX_TOP_LEVEL OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(SCATTERFIX_TOP_LEVEL ON)
    include(CTest)
endif()
option(SCATTERFIX_RECOVERY "Build image recovery using FFmpeg" ON)
option(SCATTERFIX_BUILD_CLI "Build standalone programs" ${SCATTERFIX_TOP_LEVEL})
option(SCATTERFIX_BUILD_TESTS "Build Scatterfix tests" ${SCATTERFIX_TOP_LEVEL})
set(CORE_SOURCES src/scatterfix.c src/io.c src/ts_packet.c src/ts_parser.c
    src/psi.c src/continuity.c src/ml_model.c src/repair.c src/stats.c src/contest.c)
add_library(scatterfix STATIC ${CORE_SOURCES})
add_library(scatterfix::scatterfix ALIAS scatterfix)
target_include_directories(scatterfix PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(scatterfix PUBLIC c_std_11)
if(WIN32)
    target_link_libraries(scatterfix PUBLIC ws2_32)
else()
    target_link_libraries(scatterfix PUBLIC m)
endif()
if(SCATTERFIX_RECOVERY)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavformat libavutil libswscale)
    target_sources(scatterfix PRIVATE src/recovery.c src/media.c src/rank_model.c)
    target_link_libraries(scatterfix PUBLIC PkgConfig::FFMPEG)
    add_executable(scatterfix-embedded-example EXCLUDE_FROM_ALL examples/embedded.c)
    target_link_libraries(scatterfix-embedded-example PRIVATE scatterfix)
endif()
if(SCATTERFIX_BUILD_CLI)
    if(NOT SCATTERFIX_RECOVERY)
        message(FATAL_ERROR "CLI requires SCATTERFIX_RECOVERY; disable SCATTERFIX_BUILD_CLI for a core-only library")
    endif()
    find_package(Threads REQUIRED)
    add_library(scatterfix-cli OBJECT tools/contest_cli.c)
    target_link_libraries(scatterfix-cli PRIVATE scatterfix Threads::Threads)
    add_executable(tsscatterfix src/main.c $<TARGET_OBJECTS:scatterfix-cli>)
    target_link_libraries(tsscatterfix PRIVATE scatterfix Threads::Threads)
    add_executable(scatterfix-contest tools/contest_main.c $<TARGET_OBJECTS:scatterfix-cli>)
    target_link_libraries(scatterfix-contest PRIVATE scatterfix Threads::Threads)
    add_executable(scatterfix-model tools/model.c)
    target_link_libraries(scatterfix-model PRIVATE scatterfix)
endif()
if(SCATTERFIX_BUILD_TESTS AND SCATTERFIX_RECOVERY)
    add_executable(test_scatterfix tests/test_scatterfix.c)
    target_link_libraries(test_scatterfix PRIVATE scatterfix)
    add_test(NAME scatterfix_api COMMAND test_scatterfix)
    set_tests_properties(scatterfix_api PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endif()
